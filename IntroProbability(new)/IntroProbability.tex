% Draw the pips of one die face.  The face itself is a 2 x 2 square.
% Draw the pips of one die face.  The face itself is a 2 x 2 square.
\newcommand{\drawpips}[1]{%
  \pgfmathparse{#1==2 || #1==4 || #1==5 || #1==6}%
  \ifnum\pgfmathresult>0
    \fill[pip] ( .52, .52) circle[radius=.145];
    \fill[pip] (-.52,-.52) circle[radius=.145];
  \fi
  \pgfmathparse{#1==3 || #1==4 || #1==5 || #1==6}%
  \ifnum\pgfmathresult>0
    \fill[pip] (-.52, .52) circle[radius=.145];
    \fill[pip] ( .52,-.52) circle[radius=.145];
  \fi
  \pgfmathparse{#1==1 || #1==3 || #1==5}%
  \ifnum\pgfmathresult>0
    \fill[pip] (0,0) circle[radius=.145];
  \fi
  \ifnum#1=6
    \fill[pip] ( .52,0) circle[radius=.145];
    \fill[pip] (-.52,0) circle[radius=.145];
  \fi
}

% x, y, viewing elevation, viewing azimuth, scale, top, left, right
\newcommand{\flyingdie}[8]{%
  \tdplotsetmaincoords{#3}{#4}%
  \begin{scope}[
    shift={(#1,#2)},
    tdplot_main_coords,
    scale=#5,
    line join=round,
    line cap=round,
    pip/.style={black!86!green}
  ]
    % Exact-size black backing prevents anti-aliasing gaps without protruding
    % beyond the silhouette of the die.
    \begin{scope}[canvas is xy plane at z=1]
      \fill[black] (-1,-1) rectangle (1,1);
    \end{scope}
    \begin{scope}[canvas is xz plane at y=1]
      \fill[black] (-1,-1) rectangle (1,1);
    \end{scope}
    \begin{scope}[canvas is yz plane at x=1]
      \fill[black] (-1,-1) rectangle (1,1);
    \end{scope}

    % Three visible faces, shaded separately to emphasize the rotation.
    \begin{scope}[canvas is xy plane at z=1]
      \path[draw=black,line width=.9pt,rounded corners=2pt,
            top color=white,bottom color=green!5!white]
        (-1,-1) rectangle (1,1);
      \drawpips{#6}
    \end{scope}
    \begin{scope}[canvas is xz plane at y=1]
      \path[draw=black,line width=.9pt,rounded corners=2pt,
            top color=white,bottom color=green!13!white]
        (-1,-1) rectangle (1,1);
      \drawpips{#7}
    \end{scope}
    \begin{scope}[canvas is yz plane at x=1]
      \path[draw=black,line width=.9pt,rounded corners=2pt,
            top color=green!4!white,bottom color=green!20!white]
        (-1,-1) rectangle (1,1);
      \drawpips{#8}
    \end{scope}
  \end{scope}%
}
